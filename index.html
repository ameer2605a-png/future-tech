-- Accounts Table CREATE TABLE accounts ( id SERIAL PRIMARY KEY, name VARCHAR(100) NOT NULL, account_type VARCHAR(20) NOT NULL CHECK (account_type IN ('ASSET', 'LIABILITY', 'EQUITY', 'REVENUE', 'EXPENSE')) ); -- Journal Entry Header CREATE TABLE journal_entries ( id SERIAL PRIMARY KEY, entry_date DATE NOT NULL DEFAULT CURRENT_DATE, description TEXT NOT NULL ); -- Individual Line Items (Debits and Credits) CREATE TABLE journal_lines ( id SERIAL PRIMARY KEY, journal_entry_id INT REFERENCES journal_entries(id) ON DELETE CASCADE, account_id INT REFERENCES accounts(id), line_type VARCHAR(6) CHECK (line_type IN ('DEBIT', 'CREDIT')), amount NUMERIC(15, 2) NOT NULL CHECK (amount > 0) ); -- Trigger Function: Enforce Debit = Credit Rule CREATE OR REPLACE FUNCTION verify_journal_balance() RETURNS TRIGGER AS $$ DECLARE total_debits NUMERIC(15, 2); total_credits NUMERIC(15, 2); BEGIN SELECT COALESCE(SUM(amount), 0) INTO total_debits FROM journal_lines WHERE journal_entry_id = NEW.journal_entry_id AND line_type = 'DEBIT'; SELECT COALESCE(SUM(amount), 0) INTO total_credits FROM journal_lines WHERE journal_entry_id = NEW.journal_entry_id AND line_type = 'CREDIT'; IF total_debits <> total_credits THEN RAISE EXCEPTION 'Transaction unbalanced: Total Debits (%) must equal Total Credits (%)', total_debits, total_credits; END IF; RETURN NEW; END; $$ LANGUAGE plpgsql;
